**write_block:**
```rust
fn write_block(
    &self, source_id: &str, block_hash: &[u8; 32], parent_hash: &[u8; 32],
    height: u64, events: &[(&[u8], &[u8])],
) -> Result<bool, Self::Error> {
    let conn = self.conn.lock().unwrap();

    // Block-level dedup check
    let exists: bool = conn.query_row(
        "SELECT EXISTS(SELECT 1 FROM blocks WHERE hash = ?1)",
        rusqlite::params![&block_hash[..]],
        |row| row.get(0),
    )?;
    if exists { return Ok(false); }

    let tx = conn.unchecked_transaction()?;

    // Insert block
    tx.execute(
        "INSERT INTO blocks (hash, parent_hash, height, source_id) VALUES (?1, ?2, ?3, ?4)",
        rusqlite::params![&block_hash[..], &parent_hash[..], height as i64, source_id],
    )?;

    // Insert events
    {
        let mut stmt = tx.prepare_cached(
            "INSERT INTO events (block_hash, cursor, payload) VALUES (?1, ?2, ?3)"
        )?;
        for (cursor_bytes, payload_bytes) in events {
            stmt.execute(rusqlite::params![&block_hash[..], *cursor_bytes, *payload_bytes])?;
        }
    }

    // Update canonical_tip (upsert)
    tx.execute(
        "INSERT INTO canonical_tip (source_id, block_hash) VALUES (?1, ?2)
         ON CONFLICT(source_id) DO UPDATE SET block_hash = excluded.block_hash",
        rusqlite::params![source_id, &block_hash[..]],
    )?;

    tx.commit()?;
    Ok(true)
}
```

**get_canonical_tip:**
```rust
fn get_canonical_tip(&self, source_id: &str) -> Result<Option<[u8; 32]>, Self::Error> {
    let conn = self.conn.lock().unwrap();
    let mut stmt = conn.prepare_cached(
        "SELECT block_hash FROM canonical_tip WHERE source_id = ?1"
    )?;
    let mut rows = stmt.query_map(rusqlite::params![source_id], |row| {
        let bytes: Vec<u8> = row.get(0)?;
        let mut arr = [0u8; 32];
        arr.copy_from_slice(&bytes);
        Ok(arr)
    })?;
    match rows.next() {
        Some(row) => Ok(Some(row?)),
        None => Ok(None),
    }
}
```

```rust
pub mod recovery;

use crate::cursor::ChainCursor;
use crate::block::Block;
use crate::event::Event;
use crate::machine::{ReducerCore, StateMachine};
use crate::storage::Storage;
use tokio::sync::watch;

// Codec trait (already defined in Plan 01, keep it)
pub trait Codec: Send + Sync + 'static { ... }

/// Configuration for the Engine coordinator.
pub struct EngineConfig {
    /// Logical source identifier.
    pub source_id: String,
    /// Create a snapshot after this many distinct (non-duplicate) block inserts.
    pub snapshot_every_n: usize,
}

/// Block-oriented coordinator. Owns ReducerCore + Storage + Codec.
///
/// Write-first pipeline:
/// 1. Encode block events to storage bytes via Codec
/// 2. Write block + events + update canonical_tip atomically via Storage
/// 3. Apply events to ReducerCore (state advances)
/// 4. Trigger snapshot if block insert count crosses threshold
pub struct Engine<SM, S, C>
where
    SM: StateMachine,
    S: Storage,
    C: Codec<Cursor = SM::Cursor, Event = SM::Event, State = SM::State>,
{
    core: ReducerCore<SM>,
    storage: S,
    codec: C,
    config: EngineConfig,
    /// Count of new block inserts since last snapshot.
    blocks_since_snapshot: usize,
}

impl<SM, S, C> Engine<SM, S, C>
where
    SM: StateMachine,
    S: Storage,
    C: Codec<Cursor = SM::Cursor, Event = SM::Event, State = SM::State>,
{
    pub fn new(core: ReducerCore<SM>, storage: S, codec: C, config: EngineConfig) -> Self {
        Self { core, storage, codec, config, blocks_since_snapshot: 0 }
    }

    /// Process a block through the write-first pipeline.
    /// Returns true if the block was new (inserted), false if duplicate (skipped).
    pub fn process_block(&mut self, block: &Block<SM::Cursor, SM::Event>) -> Result<bool, S::Error> {
        // Step 1: Encode events to storage bytes
        let encoded_events: Vec<(Vec<u8>, Vec<u8>)> = block.events.iter().map(|e| {
            let cursor_bytes = postcard::to_allocvec(&e.cursor)
                .expect("cursor serialization is infallible for well-formed types");
            let payload_bytes = self.codec.encode_event(&e.data);
            (cursor_bytes, payload_bytes)
        }).collect();

        let event_refs: Vec<(&[u8], &[u8])> = encoded_events.iter()
            .map(|(c, p)| (c.as_slice(), p.as_slice()))
            .collect();

        // Step 2: Write to storage (write-first invariant)
        let is_new = self.storage.write_block(
            &self.config.source_id,
            &block.block_hash,
            &block.parent_hash,
            block.height,
            &event_refs,
        )?;

        if !is_new {
            return Ok(false); // Duplicate block, skip apply
        }

        // Step 3: Apply events to ReducerCore (AFTER storage write)
        for event in &block.events {
            self.core.apply(event);
        }

        // Step 4: Snapshot triggering
        self.blocks_since_snapshot += 1;
        if self.blocks_since_snapshot >= self.config.snapshot_every_n {
            let state = self.state();
            let payload = self.codec.encode_state(&state);
            self.storage.write_snapshot(
                &self.config.source_id,
                &block.block_hash,
                &payload,
            )?;
            self.blocks_since_snapshot = 0;
        }

        Ok(true)
    }

    /// Get the current state (clones).
    pub fn state(&self) -> SM::State {
        self.core.subscribe().borrow().clone()
    }

    /// Subscribe to state changes.
    pub fn subscribe(&self) -> watch::Receiver<SM::State> {
        self.core.subscribe()
    }

    /// Get a reference to the underlying storage.
    pub fn storage(&self) -> &S {
        &self.storage
    }

    /// Get the source_id.
    pub fn source_id(&self) -> &str {
        &self.config.source_id
    }

    /// Get a reference to the codec.
    pub fn codec(&self) -> &C {
        &self.codec
    }

    /// Replace state (for recovery pipeline).
    pub fn replace_state(&mut self, state: SM::State) {
        self.core.reset_state(state);
    }

    /// Replay events from a block WITHOUT writing to storage.
    /// Used during recovery to replay stored events through ReducerCore.
    pub fn replay_events(&mut self, events: &[Event<SM::Cursor, SM::Event>]) {
        for event in events {
            self.core.apply(event);
        }
    }
}
```
