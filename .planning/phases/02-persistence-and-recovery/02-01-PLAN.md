- Define new Storage trait:

```rust
pub mod schema;

/// Block DAG storage trait. All methods are synchronous.
/// Implementations wrap with Mutex<Connection> for thread safety.
pub trait Storage: Send + Sync + 'static {
    type Error: std::error::Error + Send + Sync + 'static;

    /// Write a block (metadata + events) atomically. Returns true if the block
    /// was new (inserted), false if it was a duplicate (block_hash already exists).
    /// Updates canonical_tip to this block's hash in the same transaction.
    ///
    /// Deduplication is at block granularity: if block_hash exists, skip entirely.
    fn write_block(
        &self,
        source_id: &str,
        block_hash: &[u8; 32],
        parent_hash: &[u8; 32],
        height: u64,
        events: &[(&[u8], &[u8])],  // (cursor_bytes, payload_bytes) pairs
    ) -> Result<bool, Self::Error>;

    /// Get the canonical tip block hash for a source, or None if no blocks stored.
    fn get_canonical_tip(&self, source_id: &str) -> Result<Option<[u8; 32]>, Self::Error>;

    /// Get the height of a block by its hash, or None if the block doesn't exist.
    fn get_block_height(&self, block_hash: &[u8; 32]) -> Result<Option<u64>, Self::Error>;

    /// Write a state snapshot associated with a block hash.
    fn write_snapshot(
        &self,
        source_id: &str,
        block_hash: &[u8; 32],
        payload: &[u8],
    ) -> Result<(), Self::Error>;

    /// Walk parent pointers from tip_hash backward until a block with a snapshot
    /// is found. Returns (block_hash, snapshot_payload) or None if no snapshot exists
    /// on the canonical chain (including genesis reached).
    fn find_nearest_snapshot(
        &self,
        source_id: &str,
        tip_hash: &[u8; 32],
    ) -> Result<Option<([u8; 32], Vec<u8>)>, Self::Error>;

    /// Collect block hashes on the canonical chain from from_hash (exclusive) to
    /// to_hash (inclusive), in forward order (oldest to newest).
    /// Used for replay: collect the path, then load events for each block.
    fn collect_canonical_chain(
        &self,
        from_hash: &[u8; 32],
        to_hash: &[u8; 32],
    ) -> Result<Vec<[u8; 32]>, Self::Error>;

    /// Load events for a specific block, in insertion order.
    /// Returns (cursor_bytes, payload_bytes) pairs.
    fn load_block_events(
        &self,
        block_hash: &[u8; 32],
    ) -> Result<Vec<(Vec<u8>, Vec<u8>)>, Self::Error>;
}
```

use rusqlite_migration::{Migrations, M};

pub fn migrations() -> Migrations<'static> {
    Migrations::new(vec![
        // v1 — old linear schema (kept for migration chain)
        M::up(
            "CREATE TABLE IF NOT EXISTS events ( ... old schema ... );
             CREATE TABLE IF NOT EXISTS snapshots ( ... );
             CREATE TABLE IF NOT EXISTS anchors ( ... );
             CREATE TABLE IF NOT EXISTS coverage ( ... );
             CREATE INDEX IF NOT EXISTS ... ;"
        ),
        // v2 — block DAG schema (replaces v1 tables)
        M::up(
            "DROP TABLE IF EXISTS coverage;
             DROP TABLE IF EXISTS anchors;
             DROP TABLE IF EXISTS events;
             DROP TABLE IF EXISTS snapshots;

             CREATE TABLE blocks (
                 hash        BLOB    NOT NULL PRIMARY KEY,
                 parent_hash BLOB    NOT NULL,
                 height      INTEGER NOT NULL,
                 source_id   TEXT    NOT NULL
             );
             CREATE INDEX idx_blocks_source_height ON blocks(source_id, height);
             CREATE INDEX idx_blocks_source_parent ON blocks(source_id, parent_hash);

             CREATE TABLE events (
                 id          INTEGER PRIMARY KEY AUTOINCREMENT,
                 block_hash  BLOB    NOT NULL REFERENCES blocks(hash),
                 cursor      BLOB    NOT NULL,
                 payload     BLOB    NOT NULL
             );
             CREATE INDEX idx_events_block ON events(block_hash);

             CREATE TABLE snapshots (
                 source_id   TEXT    NOT NULL,
                 block_hash  BLOB    NOT NULL REFERENCES blocks(hash),
                 payload     BLOB    NOT NULL,
                 PRIMARY KEY (source_id, block_hash)
             );

             CREATE TABLE canonical_tip (
                 source_id   TEXT    NOT NULL PRIMARY KEY,
                 block_hash  BLOB    NOT NULL REFERENCES blocks(hash)
             );"
        ),
    ])
}
```

pub mod recovery;

use crate::cursor::ChainCursor;
use crate::event::Event;

/// Bridges typed events/state to opaque byte representation for storage.
///
/// Client provides encode/decode for event payloads and state only.
/// Cursor serialization is library-internal (postcard).
/// No cursor_to_tuple/tuple_to_cursor — cursors stored as postcard bytes.
pub trait Codec: Send + Sync + 'static {
    type Cursor: ChainCursor;
    type Event: Send + 'static;
    type State: Clone + Send + 'static;

    /// Serialize event payload to bytes.
    fn encode_event(&self, event: &Self::Event) -> Vec<u8>;

    /// Deserialize event payload from bytes. Returns None on corrupt/incompatible data.
    fn decode_event(&self, payload: &[u8]) -> Option<Self::Event>;

    /// Serialize state to opaque bytes for snapshot storage.
    fn encode_state(&self, state: &Self::State) -> Vec<u8>;

    /// Deserialize state from opaque bytes. Returns None on corrupt/incompatible.
    fn decode_state(&self, bytes: &[u8]) -> Option<Self::State>;
}
```

Note on Codec: encode_event/decode_event operate on the Event PAYLOAD (Self::Event), not the full Event<C,E>. The cursor is serialized separately by the library using postcard. This is the key simplification from the old Codec.
